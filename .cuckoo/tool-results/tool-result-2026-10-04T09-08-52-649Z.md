160: /**
161:  * 沿 parent_id 回溯出「主对话链」，取最近 ratio 比例，保证成对（USER + ASSISTANT），返回降序 id。
162:  *
163:  * 为什么不能直接按 message_id 排序取尾部：
164:  *   DeepSeek 的消息是**树**。重新生成/编辑会产生分支（一个 parent 有多个 child），
165:  *   message_id 全局递增 → 按 id 排序会把"被丢弃的分支"也混进来，角色序列不再是
166:  *   USER/ASSISTANT 交替 → share/create 报 MESSAGES_MUST_APPEAR_IN_PAIRS。
167:  *   必须从「当前叶子」沿 parent_id 回溯，得到真正的主线。
168:  *
169:  * @param msgs 全量消息（任意顺序，内部建 map）
170:  * @param leafId 当前叶子消息 id（chat_session.current_message_id），无则用最大 message_id
171:  * @param ratio 取最近比例
172:  */
173: function pickRecentPairedIds(msgs: MessageItem[], leafId: number | null, ratio: number): number[] {
174:   if (msgs.length === 0) return [];
175:   const byId = new Map<number, MessageItem>();
176:   for (const m of msgs) byId.set(m.message_id, m);
177: 
178:   // 1) 确定叶子：优先 current_message_id，否则用最大 message_id
179:   let leaf: number | null = (leafId != null && byId.has(leafId)) ? leafId : null;
180:   if (leaf == null) {
181:     let maxId: number | null = null;
182:     for (const m of msgs) if (maxId == null || m.message_id > maxId) maxId = m.message_id;
183:     leaf = maxId;
184:   }
185:   if (leaf == null) return [];
186: 
187:   // 2) 沿 parent_id 回溯主链（叶子 → 根），防环
188:   const chain: MessageItem[] = [];
189:   const seen = new Set<number>();
190:   let cur: number | null = leaf;
191:   while (cur != null && byId.has(cur) && !seen.has(cur)) {
192:     seen.add(cur);
193:     const m: MessageItem = byId.get(cur) as MessageItem;
194:     chain.push(m);
195:     cur = (typeof m.parent_id === 'number') ? (m.parent_id as number) : null;
196:   }
197:   chain.reverse(); // 根 → 叶子
198: 
199:   // 3) 回溯失败（无 parent_id 数据）→ 回退到旧的"按 id 排序取尾部"
200:   let ordered: MessageItem[];
201:   if (chain.length <= 1 && msgs.length > 1) {
202:     ordered = msgs;
203:   } else {
204:     ordered = chain;
205:   }
206: 
207:   // 4) 取最近 ratio，保证成对
208:   const total = ordered.length;
209:   let keep = Math.max(2, Math.round(total * ratio));
210:   let start = Math.max(0, total - keep);
211:   while (start > 0 && ordered[start].role !== 'USER') start--;
212:   let end = total;
213:   while (end > start + 1 && ordered[end - 1].role !== 'ASSISTANT') end--;
214:   const picked = ordered.slice(start, end);
215:   return picked.map((m) => m.message_id).sort((a, b) => b - a);
216: }
