44: /** 版本比较：a 比 b 新则返回 true。非数字段一律当 0，非法输入返回 false。 */
45: function isNewerVersion(a: any, b: any): boolean {
46:   if (!a || !b) return false;
47:   const pa = String(a).split('.').map((x) => parseInt(x, 10) || 0);
48:   const pb = String(b).split('.').map((x) => parseInt(x, 10) || 0);
49:   const n = Math.max(pa.length, pb.length);
50:   for (let i = 0; i < n; i++) {
51:     const x = pa[i] || 0, y = pb[i] || 0;
52:     if (x > y) return true;
53:     if (x < y) return false;
54:   }
55:   return false;
56: }
