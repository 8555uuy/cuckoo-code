143:       const timeoutId = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);
144: 
145:       try {
146:         const response = await fetch(input.url, {
147:           method: 'GET',
148:           signal: controller.signal,
149:         });
150: 
151:         // 流式读取，限制大小
152:         const reader = response.body ? response.body.getReader() : null;
153:         let receivedBytes = 0;
154:         const chunks: Uint8Array[] = [];
155:         let truncated = false;
156:         const MAX_BYTES = 512000;
157: 
158:         if (reader) {
159:           while (true) {
160:             const { done, value } = await reader.read();
161:             if (done) break;
162:             receivedBytes += value!.byteLength;
163:             if (receivedBytes > MAX_BYTES) {
164:               const remaining = MAX_BYTES - (receivedBytes - value!.byteLength);
165:               if (remaining > 0) chunks.push(value!.slice(0, remaining));
166:               truncated = true;
167:               try { await reader.cancel(); } catch (_) {}
168:               break;
169:             }
170:             chunks.push(value!);
171:           }
172:         }
173: 
174:         clearTimeout(timeoutId);
175: 
176:         const buffer = Buffer.concat(chunks.map(c => Buffer.from(c)));
177:         const rawText = buffer.toString('utf8');
178: 
179:         // 判断 body kind：content-type 含 html 则为 html，否则 text
180:         const contentType = response.headers.get('content-type') || '';
