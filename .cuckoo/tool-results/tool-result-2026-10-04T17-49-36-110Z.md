105: function safeEntryFilter(entryPath: string, entry: any): boolean {
106:   if (!entryPath) return false;
107:   if (entryPath.startsWith('/') || entryPath.startsWith('\\')) return false;
108:   if (/^[A-Za-z]:/.test(entryPath)) return false;
109:   const normalized = entryPath.replace(/\\/g, '/');
110:   if (normalized.split('/').includes('..')) return false;
111:   const type = entry && entry.type;
112:   if (type === 'SymbolicLink' || type === 'Link') return false;
113:   return true;
114: }
