244: function readManifestCache(file: string): ManifestCache | null {
245:   try {
246:     if (!fs.existsSync(file)) return null;
247:     const raw = JSON.parse(fs.readFileSync(file, 'utf-8'));
248:     if (!raw || typeof raw !== 'object' || !raw.entries || typeof raw.entries !== 'object') return null;
249:     if (typeof raw.cachedAt !== 'string') return null;
250:     return { cachedAt: raw.cachedAt, entries: raw.entries };
251:   } catch {
252:     return null;
253:   }

256: /** 带并发上限的 map，保持输入顺序 */
257: async function mapWithConcurrency<T, R>(items: T[], limit: number, fn: (item: T) => Promise<R>): Promise<R[]> {
258:   const results = new Array<R>(items.length);
259:   let cursor = 0;
260:   const workerCount = Math.max(1, Math.min(limit, items.length));
261:   const workers: Promise<void>[] = [];
262:   for (let w = 0; w < workerCount; w++) {
263:     workers.push((async () => {
264:       for (;;) {
265:         const i = cursor++;
266:         if (i >= items.length) return;
267:         results[i] = await fn(items[i]);
268:       }
269:     })());
270:   }
271:   await Promise.all(workers);
272:   return results;
273: }

322: /**
