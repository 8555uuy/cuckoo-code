4:  * 单独成文件：安装器（下载 tarball）与市场（拉远端 plugin.json）都要用同一套
5:  * repo/branch 校验。若只放在 installer 里，市场为了复用就会把 `tar` 拖进模块图。
6:  *
7:  * 所有拼进 URL 的片段都必须先过校验 —— 这里是从外部输入到网络请求的唯一关口。
8:  */
9: 
10: /**
11:  * owner 校验：GitHub 用户名只允许字母数字与连字符，且首字符必须是字母数字。
12:  *
13:  * **不能**写成 `[A-Za-z0-9._-]+` —— 那样 `..` 会通过校验，`../evil` 就成了"合法仓库"，
14:  * 拼出的 URL 会被规范化成完全不同的路径。首字符限定字母数字即排除 `.` / `..`。
15:  */
16: const OWNER_RE = /^[A-Za-z0-9][A-Za-z0-9-]{0,38}$/;
17: 
18: /** 仓库名校验：允许字母数字与 . _ - */
19: const REPO_NAME_RE = /^[A-Za-z0-9._-]{1,100}$/;
20: 
21: /** 分支名分段：允许字母数字与 . _ - */
22: const BRANCH_SEG_RE = /^[A-Za-z0-9._-]+$/;
23: 
24: /** tarball 下载通道（不消耗 GitHub API 配额） */
25: const CODELOAD_BASE = 'https://codeload.github.com';
26: 
27: /** 原始文件通道（CDN，不消耗 API 配额） */
28: const RAW_BASE = 'https://raw.githubusercontent.com';
29: 
30: /** owner/repo 校验：两段，owner 严格、仓库名显式排除 `.` 与 `..` */
31: function isValidRepo(repo: unknown): repo is string {
32:   if (typeof repo !== 'string') return false;
33:   const parts = repo.split('/');
34:   if (parts.length !== 2) return false;
35:   const [owner, name] = parts;
36:   if (!OWNER_RE.test(owner)) return false;
37:   if (name === '.' || name === '..') return false;
38:   return REPO_NAME_RE.test(name);
39: }
40: 
41: /**
42:  * 分支名校验：逐段检查，拒绝空段与 `..`。
43:  * 分支名会被拼进 URL 路径，不能放任任意字符。
44:  */
45: function isValidBranch(branch: unknown): branch is string {
46:   if (typeof branch !== 'string' || !branch.trim()) return false;
47:   const segs = branch.split('/');
48:   for (const s of segs) {
49:     if (!s) return false;
50:     if (s === '.' || s === '..') return false;
51:     if (!BRANCH_SEG_RE.test(s)) return false;
52:   }
53:   return true;
54: }
55: 
56: /** tarball 下载地址（安装用） */
57: function buildTarballUrl(repo: string, branch: string): string {
58:   return CODELOAD_BASE + '/' + repo + '/tar.gz/refs/heads/' + branch;
59: }
60: 
61: /** 仓库根目录下某个文件的 raw 地址（读远端 plugin.json 用） */
62: function buildRawFileUrl(repo: string, branch: string, file: string): string {
63:   return RAW_BASE + '/' + repo + '/' + branch + '/' + file;
64: }
