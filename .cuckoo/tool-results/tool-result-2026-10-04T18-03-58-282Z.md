2:  * 子代理窗口的 bridge 逻辑（仅在子代理窗口中激活）
3:  *
4:  * 激活条件：主进程经 --cuckoo-subagent=<encoded JSON> 传入配置。
5:  * 流程：页面就绪 → 发任务提示词 → 监听回复 → 无工具调用（纯文本）= 完成

24: /** 从 process.argv 解析子代理配置（无则 null） */
25: export function readSubagentConfig(): SubagentConfig | null {
26:   try {
27:     const argv = (process as any).argv || [];
28:     const arg = argv.find((a: string) => a.startsWith('--cuckoo-subagent='));

30:     const json = decodeURIComponent(arg.slice('--cuckoo-subagent='.length));
31:     const cfg = JSON.parse(json);
